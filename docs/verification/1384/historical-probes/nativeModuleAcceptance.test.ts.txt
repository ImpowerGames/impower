const moduleA = `
        --!strict
        local module = {}

        function module.foo()
            return 2
        end

        function module.bar()
            local m = require(game.B)
            return m.foo() + 1
        end

        return module
    `;
const moduleB = `
        --!strict
        local module = {}

        function module.foo()
            return 2
        end

        function module.bar()
            local m = require(game.A)
            return m.foo() + 1
        end

        return module
    `;

const originalCase: PortedCase = {
  name: "cycles_dont_make_everything_any",
  fixture: "BuiltinsFixture",
  source: moduleA,
  module: "game/A",
  // Registration does not perform check(B). A is supplied by the entry source.
  moduleSources: { "game/B": moduleB },
  // Sole original assertion. Keep actual diagnostics; upstream asserts no count.
  expect: [{ moduleReturn: true, module: "game/B", equals: "module" }],
};

// No case-body cycle flag override: the shared all-Luau-flags-on configuration

beforeEach(() => vi.stubEnv("LUAU_TYPECHECK_AREAS", "TypeInfer.modules.test.cpp"));

it(originalCase.name, async () => {

  expect(sha(moduleA)).toBe("c6ad5cd2e10cdfac72dd2a3759bc65b8aac8c44445f53cad9843c6af8888c974");
  expect(sha(moduleB)).toBe("ca2edcf866d1fd36bad23ca054cb8f0dc81e191763490239b8753a5edfcfb95a");
  await runNativePortedCase("TypeInfer.modules.test.cpp", originalCase, required);
});
