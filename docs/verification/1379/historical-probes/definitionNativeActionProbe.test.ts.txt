/** C++ raw-literal bytes after phase-one line-ending normalization, checked against the retained exact-pin audit. */
function source(name: string, index = 0, area = "definitions"): string {
  const file = area === "definitions" ? definitions : metatable;
  const marker = `TEST_CASE_FIXTURE(Fixture, "${name}")`;
  const start = file.indexOf(marker);
  expect(start, `Missing exact original ${name}`).toBeGreaterThanOrEqual(0);
  const next = file.indexOf("TEST_CASE_FIXTURE(", start + marker.length);
  const body = file.slice(start, next < 0 ? undefined : next);
  const literals = [...body.matchAll(/R"\((.*?)\)"/gs)];
  expect(literals[index], `Missing original raw literal ${name}:${index}`).toBeDefined();
  const value = literals[index]![1]!;
  const prior = audited.find(row => row.area === area && row.name === name && row.index === index);
  expect(prior, `Missing retained exact-pin source evidence ${name}:${index}`).toBeDefined();
  expect(value).toBe(prior!.source);
  expect(sha(value)).toBe(prior!.sourceSha256);
  observations[`source:${name}:${index}`] = { sha256: sha(value), bytes: Buffer.byteLength(value) };
  return value;
}

function direct(name: string, index = 0) {
  const actual = fixture.observe("definition", source(name, index));
  observations[`load:${name}:${index}`] = actual;
  expect(actual.status).toBe("ok");
  return actual;
}

function missingBinding(name: string) {
  const context = fixture.context();
  const actual = fixture.observe("binding_facts", context.session, context.revision, name);
  observations[`absent:${name}`] = actual;
  expect(actual).toEqual({ status: "error", message: `Missing native global binding: ${name}` });
}
