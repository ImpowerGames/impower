async function tracked(file: string, record: PortedCase, definitionOnly = false) {
  const events: unknown[] = [], skip = vi.fn();
  let native: NativeFixture | undefined;
  const load = vi.fn(async () => {
    native = await loadNativeFixture();
    const fixture = native, observe = fixture.observe.bind(fixture);
    vi.spyOn(fixture, "observe").mockImplementation((operation, ...args) => {
      const actual = observe(operation, ...args);
      if (["create", "flag", "definition", "select_new_solver", "source", "check", "dispose"].includes(operation))
        events.push({ operation, args: args.map(arg => typeof arg === "string" && arg.length > 100 ? { sha256: sha(arg) } : arg), actual });
      return actual;
    });
    const check = fixture.check.bind(fixture);
    vi.spyOn(fixture, "check").mockImplementation((...args) => {
      if (definitionOnly) throw Error("Unexpected source check in original definition-only case");
      const actual = check(...args);
      events.push({ checked: args[0], diagnostics: actual.diagnostics, flags: actual.effectiveFlags, graph: fixture.moduleFacts(actual, args[0]) });
      return actual;
    });
    return fixture;
  });
  try {
    await runNativePortedCase(file, record, skip, load);
  } finally {
    receipt[`dispatch:${file}:${record.name}`] = events;
    if (native) expect(() => native!.heapBytes()).toThrow("Disposed native fixture host");
  }
  expect(load).toHaveBeenCalledTimes(1); expect(skip).not.toHaveBeenCalled();
  expect(events.filter(event => (event as { operation?: string }).operation === "create")).toHaveLength(1);
  expect(events.filter(event => (event as { operation?: string }).operation === "dispose")).toHaveLength(1);
  return events;
}
